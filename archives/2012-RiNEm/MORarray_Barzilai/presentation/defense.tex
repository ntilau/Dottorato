\subsection{Near field to far field transformation}
\frame
{
\frametitle{Near field to far field transformation}
\begin{itemize}
\item \alert{Huygens' principle}: far field derived from near field $\tilde{\mbfit{E}}=\tilde{\mbfit{E}}(\mbfit{r'})$ e $\tilde{\mbfit{H}}=\tilde{\mbfit{H}}(\mbfit{r'})$, $\mbfit{r'} \in \mrm{\Gamma_R}$
% dalle correnti equivalenti} ${\mbfit{J}}_s^{eq} = \hat{\mbfit{n}} \times \tilde{\mbfit{H}}$ e ${\mbfit{J}}_{ms}^{eq} = - \hat{\mbfit{n}} \times \tilde{\mbfit{E}}$ calcolate su $\mrm{\Gamma_R}$\\[.2cm]
%\hspace{-.5cm}
\begin{center}
\vspace{-.5cm} \scriptsize $$\hspace{-.5cm}\check{\mbfit{E}}(\theta, \phi) = j \frac{\zeta_0 k_0}{4 \pi} \int \! \! \! \! \int_\mrm{\Gamma_R}  \mrm{e}^{jk_0 \hat{\mbfit{r}} \cdot \mbfit{r'}} \left [ - (\hat{\mbfit{n}} \times \tilde{\mbfit{H}})  + \left ( ( \hat{\mbfit{n}} \times \tilde{\mbfit{H}} ) \cdot \hat{\mbfit{r}} \right ) \hat{\mbfit{r}} + \frac{1}{\zeta_0} \ (\hat{\mbfit{n}} \times \tilde{\mbfit{E}})  \times \hat{\mbfit{r}} \right ]  d{\Gamma}$$ \vspace{-.2cm} \footnotesize $$ \hat{\mbfit{r}} = \hat{\mbfit{r}} \left ( \theta, \phi \right ) \textnormal{ unit vector in  the \alert{look angle direction}}$$ \vspace{-.5cm} $$\hat{\mbfit{n}} \textnormal{ normal on } \mrm{\Gamma_R} \textnormal{ outwardly directed from } \mrm{\Omega}_h $$
\end{center} %  \\[.3cm] 
%\vspace{.2cm} 
\pause
\item \normalsize \alert{Radiated field approximation}: $\check{\mbfit{H}}(\theta, \phi) =  \frac{1}{\zeta_0} \hat{\mbfit{r}} \times \check{\mbfit{E}}(\theta, \phi )$ \\ \pause % ci permette di trascurare il campo magnetico 
%\vspace{.2cm}
%\pause
%\item \alert{Potenza radiata}: \vspace{-.2cm} $$\mrm{P}_\mrm{rad} = \frac{1}{2} \Re \left \{ \int_{\mrm{\Gamma_R}} \left [ \tilde{\mbfit{E}} \times \tilde{\mbfit{H}}^{*} \right ] \cdot \hat{\mbfit{n}} \ d\mrm{\Gamma} \right \}  = \mrm{P}_\mrm{in} - (\mrm{P}_\mrm{rifl} + \mrm{P}_\mrm{diss})$$ \\ \pause
\item \alert{Radiation pattern}: \vspace{-.2cm}
$$\mrm{G}_{\hat{\theta}} \left ( \theta, \phi \right ) = 4 \pi \frac{\left | {{\hat{\theta}}} \cdot \check{\mbfit E} (\theta, \phi) \right |^2}{ 2 \zeta_0 \mrm{P}_\mrm{acc}}, \ \mrm{G}_{\hat{\phi}} \left ( \theta, \phi \right ) = 4 \pi \frac{\left | {\hat{\phi}} \cdot \check{\mbfit E} (\theta , \phi ) \right |^2}{ 2 \zeta_0 \mrm{P}_\mrm{acc}}$$ \\[-.2cm]
$$\mrm{G}_{tot} \left ( \theta, \phi \right )  = \mrm{G}_{\hat{\theta}} \left ( \theta, \phi \right ) + \mrm{G}_{\hat{\phi}} \left ( \theta, \phi \right ) $$
\vspace{-.2cm} \footnotesize $$\mrm{P_{acc}} = \mrm{P_{inc}} - \mrm{P_{refl}}  \textnormal{: Accepted power at the feeding ports}$$ 
\end{itemize}
